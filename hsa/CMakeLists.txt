cmake_minimum_required(VERSION 3.20)
project(MacAMDGPUHSA LANGUAGES C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "The IOKit transport requires macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(HSA_RUNTIME_SOURCES src/runtime.cpp src/memory.cpp src/queues.cpp src/device_init.cpp src/isa.cpp src/code_object.cpp src/executable.cpp src/virtual_memory.cpp src/host_services.cpp src/ipc_memory.cpp src/ipc_signal.cpp src/platform_extensions.cpp)
add_library(hsa-runtime64 SHARED ${HSA_RUNTIME_SOURCES} src/transport_iokit.cpp)
target_compile_definitions(hsa-runtime64 PRIVATE HSA_EXPORT)
target_include_directories(hsa-runtime64 PUBLIC include third_party/hsa/include)
target_compile_options(hsa-runtime64 PRIVATE -Wall -Wextra -Werror)
target_link_libraries(hsa-runtime64 PRIVATE "-framework IOKit" "-framework CoreFoundation")
set_target_properties(hsa-runtime64 PROPERTIES VERSION 0.1.0 SOVERSION 1)
add_executable(mac-hsa-info tools/mac_hsa_info.c)
target_link_libraries(mac-hsa-info PRIVATE hsa-runtime64)
target_compile_options(mac-hsa-info PRIVATE -Wall -Wextra -Werror)

add_executable(mac-hsa-memory-test tools/mac_hsa_memory_test.c)
target_link_libraries(mac-hsa-memory-test PRIVATE hsa-runtime64)
target_compile_options(mac-hsa-memory-test PRIVATE -Wall -Wextra -Werror)
add_executable(mac-hsa-executable-test tools/mac_hsa_executable_test.c)
target_link_libraries(mac-hsa-executable-test PRIVATE hsa-runtime64)
target_compile_options(mac-hsa-executable-test PRIVATE -Wall -Wextra -Werror)
add_executable(mac-hsa-ipc-test tools/mac_hsa_ipc_test.c)
target_link_libraries(mac-hsa-ipc-test PRIVATE hsa-runtime64)
target_compile_options(mac-hsa-ipc-test PRIVATE -Wall -Wextra -Werror)

add_executable(mac-hsa-shared-test tools/mac_hsa_shared_test.cpp src/transport_iokit.cpp src/device_init.cpp)
target_include_directories(mac-hsa-shared-test PRIVATE src third_party/hsa/include)
target_compile_options(mac-hsa-shared-test PRIVATE -Wall -Wextra -Werror)
