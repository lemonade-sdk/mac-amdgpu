cmake_minimum_required(VERSION 3.20)
project(MacAMDGPUHSA LANGUAGES C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "The IOKit transport requires macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_library(hsa-runtime64 SHARED src/runtime.cpp src/transport_iokit.cpp)
target_compile_definitions(hsa-runtime64 PRIVATE HSA_EXPORT)
target_include_directories(hsa-runtime64 PUBLIC include third_party/hsa/include)
target_compile_options(hsa-runtime64 PRIVATE -Wall -Wextra -Werror)
target_link_libraries(hsa-runtime64 PRIVATE "-framework IOKit" "-framework CoreFoundation")
set_target_properties(hsa-runtime64 PROPERTIES VERSION 0.1.0 SOVERSION 1)
add_executable(mac-hsa-info tools/mac_hsa_info.c)
target_link_libraries(mac-hsa-info PRIVATE hsa-runtime64)
target_compile_options(mac-hsa-info PRIVATE -Wall -Wextra -Werror)

include(CTest)
if(BUILD_TESTING)
  add_executable(hsa-runtime-test ../tests/hsa_runtime_test.cpp src/runtime.cpp)
  target_include_directories(hsa-runtime-test PRIVATE include src third_party/hsa/include)
  target_compile_options(hsa-runtime-test PRIVATE -Wall -Wextra -Werror -UNDEBUG
    -fsanitize=address,undefined -fno-sanitize-recover=all)
  target_link_options(hsa-runtime-test PRIVATE -fsanitize=address,undefined)
  add_test(NAME hsa-runtime-lifecycle COMMAND hsa-runtime-test)
endif()
