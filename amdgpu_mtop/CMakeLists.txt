cmake_minimum_required(VERSION 3.20)
project(amdgpu_mtop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT APPLE)
  message(FATAL_ERROR "amdgpu_mtop requires macOS IOKit")
endif()
add_executable(amdgpu_mtop main.cpp transport.cpp)
target_compile_options(amdgpu_mtop PRIVATE -Wall -Wextra -Werror)
target_link_libraries(amdgpu_mtop PRIVATE "-framework IOKit" "-framework CoreFoundation")
include(CTest)
if(BUILD_TESTING)
  add_executable(mtop-selection-test ../tests/mtop_selection_test.cpp)
  target_compile_options(mtop-selection-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME mtop-selection COMMAND mtop-selection-test)
  add_test(NAME mtop-help COMMAND amdgpu_mtop --help)
endif()
